# 2) حافظه و ذخیره‌سازی در وب

## 2.1) مسئله دقیقاً چیست؟

در یک برنامه وب، داده‌ها معمولاً یکی از این نیازها را دارند:

+ فقط تا وقتی صفحه باز است نگه داشته شوند.

+ تا وقتی تب باز است باقی بمانند.

+ بعد از بستن مرورگر هم حفظ شوند.

+ همراه درخواست‌ها به سرور ارسال شوند.

نوع ذخیره‌سازی را با توجه به همین نیاز انتخاب می‌کنیم.

## 2.2) مقایسه خلاصه

| روش              |     ماندگاری | ارسال خودکار به سرور | نوع داده        | کاربرد رایج                   |
| ---------------- | ------------:| --------------------:| --------------- | ----------------------------- |
| RAM (متغیر JS)   | تا رفرش صفحه |                  خیر | هر نوع JS       | state موقت UI                 |
| `sessionStorage` |   تا بستن تب |                  خیر | فقط string      | draft موقت، فرم چندمرحله‌ای   |
| `localStorage`   |       پایدار |                  خیر | فقط string      | تنظیمات، کش کوچک              |
| Cookie           |   قابل تنظیم |                  بله | string (کم‌حجم) | احراز هویت، سشن، تنظیمات کوچک |
| IndexedDB        |       پایدار |                  خیر | ساختارمند       | داده حجیم و آفلاین            |

> نکته: `localStorage/sessionStorage` فقط string نگه می‌دارند. برای آبجکت/آرایه باید JSON استفاده شود.

# 3) localStorage

## 3.1) تعریف و ویژگی‌ها

`localStorage` یک فضای ذخیره‌سازی Key-Value داخل مرورگر است که:

+ داده را به صورت پایدار نگه می‌دارد (پس از بستن مرورگر هم باقی می‌ماند).

+ مقدارها را فقط به صورت **string** ذخیره می‌کند.

+ همراه درخواست‌ها به سرور ارسال نمی‌شود.

## 3.2) توابع اصلی

```js
localStorage.setItem('key', 'value')
localStorage.getItem('key')
localStorage.removeItem('key')
localStorage.clear()
localStorage.key(0)
localStorage.length
```

## 3.3) مثال‌های کاربردی

### مثال 1: ذخیره‌ی تم

```html
<button id="toggleTheme">Toggle Theme</button>
```

```js
const btn = document.querySelector('#toggleTheme');

function applyTheme(theme) {
  document.documentElement.dataset.theme = theme;
}

const savedTheme = localStorage.getItem('theme') || 'light';
applyTheme(savedTheme);

btn.addEventListener('click', () => {
  const current = localStorage.getItem('theme') || 'light';
  const next = current === 'light' ? 'dark' : 'light';
  localStorage.setItem('theme', next);
  applyTheme(next);
});
```

### مثال 2: کش ساده‌ی پاسخ API

```js
async function getProducts() {
  const key = 'products_cache_v1';
  const cached = localStorage.getItem(key);

  if (cached) {
    try {
      return JSON.parse(cached);
    } catch {
      localStorage.removeItem(key);
    }
  }

  const res = await fetch('/api/products');
  const data = await res.json();

  localStorage.setItem(key, JSON.stringify(data));
  return data;
}
```

## 3.4) مثال کامل: کش با زمان انقضا (TTL)

```js
function setCache(key, value, ttlMs) {
  const payload = {
    value,
    expiresAt: Date.now() + ttlMs
  };
  localStorage.setItem(key, JSON.stringify(payload));
}

function getCache(key) {
  const raw = localStorage.getItem(key);
  if (!raw) return null;

  try {
    const payload = JSON.parse(raw);
    if (Date.now() > payload.expiresAt) {
      localStorage.removeItem(key);
      return null;
    }
    return payload.value;
  } catch {
    localStorage.removeItem(key);
    return null;
  }
}

setCache('profile', { name: 'Shayan' }, 60_000);
console.log(getCache('profile'));
```

```js
const STORAGE_KEY = 'todos_v1';

function loadTodos() {
  const raw = localStorage.getItem(STORAGE_KEY);
  if (!raw) return [];
  try {
    return JSON.parse(raw);
  } catch {
    localStorage.removeItem(STORAGE_KEY);
    return [];
  }
}

function saveTodos(todos) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(todos));
}

let todos = loadTodos();

const form = document.querySelector('#todoForm');
const input = document.querySelector('#todoInput');
const list = document.querySelector('#todoList');

function render() {
  list.innerHTML = '';

  for (const t of todos) {
    const li = document.createElement('li');
    li.className = 'item' + (t.done ? ' done' : '');

    const left = document.createElement('div');

    const checkbox = document.createElement('input');
    checkbox.type = 'checkbox';
    checkbox.checked = t.done;
    checkbox.addEventListener('change', () => {
      t.done = checkbox.checked;
      saveTodos(todos);
      render();
    });

    const title = document.createElement('span');
    title.className = 'title';
    title.textContent = t.title;

    left.appendChild(checkbox);
    left.appendChild(title);
    left.style.display = 'flex';
    left.style.gap = '10px';

    const actions = document.createElement('div');
    actions.className = 'actions';

    const delBtn = document.createElement('button');
    delBtn.textContent = 'حذف';
    delBtn.addEventListener('click', () => {
      todos = todos.filter(x => x.id !== t.id);
      saveTodos(todos);
      render();
    });

    actions.appendChild(delBtn);

    li.appendChild(left);
    li.appendChild(actions);

    list.appendChild(li);
  }
}

form.addEventListener('submit', (e) => {
  e.preventDefault();
  const title = input.value.trim();
  if (!title) return;

  const newTodo = {
    id: crypto.randomUUID(),
    title,
    done: false,
    createdAt: Date.now()
  };

  todos.unshift(newTodo);
  saveTodos(todos);
  input.value = '';
  render();
});

render();
```

# 4) sessionStorage

## 4.1) تعریف و تفاوت با localStorage

`sessionStorage` همان الگوی Key-Value را دارد و فقط string نگه می‌دارد؛ تفاوت اصلی این است که داده فقط تا زمانی که **همان تب** باز است باقی می‌ماند.

## 4.2) مثال‌ها

### مثال 1: draft فرم چندمرحله‌ای

```js
const KEY = 'signup_draft';

function saveDraft(data) {
  sessionStorage.setItem(KEY, JSON.stringify(data));
}

function loadDraft() {
  const raw = sessionStorage.getItem(KEY);
  return raw ? JSON.parse(raw) : null;
}
```

### مثال 2: جلوگیری از ارسال تکراری

```js
async function safeSubmit(formData) {
  const lockKey = 'submit_lock';
  if (sessionStorage.getItem(lockKey) === '1') {
    alert('در حال ارسال...');
    return;
  }

  sessionStorage.setItem(lockKey, '1');

  try {
    const res = await fetch('/api/submit', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(formData)
    });
    if (!res.ok) throw new Error('Submit failed');
    alert('ارسال شد');
  } catch {
    alert('خطا');
  } finally {
    sessionStorage.removeItem(lockKey);
  }
}
```

# 5) Cookie

## 5.1) تعریف

ویژگی Cookie مقدار کوتاهی از داده است که مرورگر نگه می‌دارد و معمولاً در درخواست‌های بعدی برای همان دامنه ارسال می‌کند.

این ویژگی باعث می‌شود کوکی برای احراز هویت، سشن و بعضی تنظیمات کوچک مناسب باشد.

## 5.2) کار با کوکی در مرورگر

### تنظیم

```
document.cookie = 'theme=dark'
```

### خواندن

```js
function getCookie(name) {
  const all = document.cookie.split('; ').map(x => x.split('='));
  const found = all.find(([k]) => k === name);
  return found ? decodeURIComponent(found[1]) : null;
}
```

### تنظیم با تاریخ انقضا

```js
function setCookie(name, value, days) {
  const expires = new Date(Date.now() + days*24*60*60*1000).toUTCString();
  document.cookie = `${name}=${encodeURIComponent(value)}; Expires=${expires}; Path=/`;
}
```

## 5.3) گزینه‌های مهم کوکی

+ `HttpOnly`: دسترسی جاوااسکریپت به کوکی را قطع می‌کند.

+ `Secure`: فقط روی HTTPS ارسال می‌شود.

+ `SameSite`: رفتار ارسال کوکی بین سایت‌ها را کنترل می‌کند.

+ `Path`: محدوده مسیر.

+ `Max-Age/Expires`: مدت اعتبار.

## 5.4) تنظیم کوکی از سمت Express

```js
import express from 'express';
import cookieParser from 'cookie-parser';

const app = express();
app.use(cookieParser());

app.get('/set', (req, res) => {
  res.cookie('theme', 'dark', {
    httpOnly: false,
    sameSite: 'lax',
    maxAge: 7 * 24 * 60 * 60 * 1000,
    path: '/'
  });
  res.json({ ok: true });
});

app.get('/read', (req, res) => {
  res.json({ cookies: req.cookies });
});

app.listen(3000);
```

# 7) JSON

## 7.1) JSON چیست؟

JSON یک فرمت متنی برای نمایش داده‌های ساختاری است. در عمل:

+ وقتی می‌گوییم «JSON»، منظور یک **string** با قواعد مشخص است.

+ برای تبدیل بین آبجکت جاوااسکریپت و JSON از `JSON.stringify` و `JSON.parse` استفاده می‌کنیم.

### قواعد مهم

+ کلیدها باید داخل کوتیشن دابل باشند.

+ کامنت ندارد.

+ `undefined` و تابع را پشتیبانی نمی‌کند.

## 7.2) `JSON.stringify`

```js
const obj = { a: 1, b: true };
const s = JSON.stringify(obj);
```

امضا:

```js
JSON.stringify(value, replacer, space)
```

### `space` برای خوانایی

```js
console.log(JSON.stringify({a:1,b:2}, null, 2));
```

### `replacer` به شکل آرایه

```js
const user = { id: 1, name: 'Ali', password: '1234' };
console.log(JSON.stringify(user, ['id', 'name']));
```

### `replacer` به شکل تابع

```js
const obj = { a: 1, b: 2, secret: 'x' };
const s = JSON.stringify(obj, (key, value) => {
  if (key === 'secret') return undefined;
  return value;
});
console.log(s);
```

## 7.3) `JSON.parse`

```js
const s = '{"a":1,"b":true}';
const obj = JSON.parse(s);
```

امضا:

```js
JSON.parse(text, reviver)
```

### `reviver` برای تبدیل مقادیر

```js
const raw = '{"createdAt":"2025-12-29T12:00:00.000Z"}';
const obj = JSON.parse(raw, (key, value) => {
  if (key === 'createdAt') return new Date(value);
  return value;
});
```

## 7.4) خطاها و نکات رایج

### Date

در JSON به string تبدیل می‌شود و هنگام parse باید برگردانده شود.

### BigInt

```js
JSON.stringify({ n: 10n }) // TypeError
```

```js
const s = JSON.stringify({ n: 10n }, (k,v) => typeof v === 'bigint' ? v.toString() : v);
```

### Circular Reference

```js
const a = {};
a.self = a;
JSON.stringify(a) // خطا
```

### Deep Clone

JSON برای clone کامل مناسب نیست. در مرورگرهای جدید می‌توان از `structuredClone` استفاده کرد:

```js
const copy = structuredClone(original)
```

## 7.5) مثال کامل: ذخیره نسخه‌دار

```js
const STORAGE_KEY = 'todos_v2';
const STORAGE_VERSION = 2;

function saveTodosV2(todos) {
  const payload = {
    version: STORAGE_VERSION,
    data: todos,
    savedAt: Date.now()
  };
  localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
}

function loadTodosV2() {
  const raw = localStorage.getItem(STORAGE_KEY);
  if (!raw) return [];

  try {
    const payload = JSON.parse(raw);

    if (payload.version === 2 && Array.isArray(payload.data)) {
      return payload.data;
    }

    if (payload.version === 1 && Array.isArray(payload.data)) {
      return payload.data.map(t => ({ ...t, priority: 1 }));
    }

    return [];
  } catch {
    localStorage.removeItem(STORAGE_KEY);
    return [];
  }
}
```

# 8) JWT

## 8.1) JWT چیست؟

JWT یک رشته‌ی امضاشده است که سرور تولید می‌کند و کلاینت در درخواست‌های بعدی آن را ارائه می‌دهد تا سرور کاربر را شناسایی کند.

## 8.2) ساختار

JWT سه بخش دارد:

```js
HEADER.PAYLOAD.SIGNATURE
```

+ Header: نوع و الگوریتم

+ Payload: داده‌ها (claims)

+ Signature: امضا

> Payload رمزگذاری نمی‌شود؛ فقط Base64Url است. داده‌ی محرمانه داخل payload قرار ندهید.

## 8.3) ذخیره توکن

دو روش رایج:

+ نگهداری توکن در حافظه/Storage و ارسال با header

+ نگهداری در کوکی `HttpOnly` و ارسال خودکار همراه درخواست

در این جزوه: توکن در **کوکی HttpOnly** ذخیره می‌شود.

## 8.4) پیاده‌سازی اولیه JWT در Express (بدون DB)

`server.js`:

```js
import express from 'express';
import cors from 'cors';
import cookieParser from 'cookie-parser';
import jwt from 'jsonwebtoken';
import bcrypt from 'bcrypt';
import dotenv from 'dotenv';

dotenv.config();

const app = express();

app.use(cors({
  origin: 'http://localhost:3000',
  credentials: true
}));
app.use(express.json());
app.use(cookieParser());
app.use(express.static('public'));

const JWT_SECRET = process.env.JWT_SECRET || 'dev_secret';

const users = []; // {id, email, passHash}

function signAccessToken(user) {
  return jwt.sign(
    { sub: user.id, email: user.email },
    JWT_SECRET,
    { expiresIn: '15m' }
  );
}

function auth(req, res, next) {
  const token = req.cookies?.access_token;
  if (!token) return res.status(401).json({ message: 'Unauthenticated' });

  try {
    const payload = jwt.verify(token, JWT_SECRET);
    req.user = { id: payload.sub, email: payload.email };
    next();
  } catch {
    return res.status(401).json({ message: 'Invalid/Expired token' });
  }
}

app.post('/api/auth/register', async (req, res) => {
  const { email, password } = req.body;
  if (!email || !password) return res.status(400).json({ message: 'email/password required' });
  if (users.some(u => u.email === email)) return res.status(409).json({ message: 'Email exists' });

  const passHash = await bcrypt.hash(password, 10);
  const user = { id: crypto.randomUUID(), email, passHash };
  users.push(user);

  res.json({ ok: true });
});

app.post('/api/auth/login', async (req, res) => {
  const { email, password } = req.body;
  const user = users.find(u => u.email === email);
  if (!user) return res.status(401).json({ message: 'Invalid credentials' });

  const ok = await bcrypt.compare(password, user.passHash);
  if (!ok) return res.status(401).json({ message: 'Invalid credentials' });

  const token = signAccessToken(user);

  res.cookie('access_token', token, {
    httpOnly: true,
    sameSite: 'lax',
    secure: false,
    maxAge: 15 * 60 * 1000
  });

  res.json({ ok: true });
});

app.post('/api/auth/logout', (req, res) => {
  res.clearCookie('access_token');
  res.json({ ok: true });
});

app.get('/api/me', auth, (req, res) => {
  res.json({ user: req.user });
});

app.listen(3000, () => console.log('http://localhost:3000'));
```

```js
import bcrypt from 'bcrypt';
import jwt from 'jsonwebtoken';
import { pool } from '../db/pool.js';

function signAccessToken(user) {
  return jwt.sign(
    { sub: user.id, email: user.email },
    process.env.JWT_SECRET,
    { expiresIn: '15m' }
  );
}

export async function register(req, res) {
  const { email, password } = req.body;
  if (!email || !password) return res.status(400).json({ message: 'email/password required' });

  const [rows] = await pool.query('SELECT id FROM users WHERE email=?', [email]);
  if (rows.length) return res.status(409).json({ message: 'Email exists' });

  const passHash = await bcrypt.hash(password, 10);
  const id = crypto.randomUUID();

  await pool.query('INSERT INTO users (id, email, pass_hash) VALUES (?, ?, ?)', [id, email, passHash]);
  res.json({ ok: true });
}

export async function login(req, res) {
  const { email, password } = req.body;
  if (!email || !password) return res.status(400).json({ message: 'email/password required' });

  const [rows] = await pool.query('SELECT id, email, pass_hash FROM users WHERE email=?', [email]);
  if (!rows.length) return res.status(401).json({ message: 'Invalid credentials' });

  const user = rows[0];
  const ok = await bcrypt.compare(password, user.pass_hash);
  if (!ok) return res.status(401).json({ message: 'Invalid credentials' });

  const token = signAccessToken({ id: user.id, email: user.email });
  const secure = process.env.COOKIE_SECURE === 'true';

  res.cookie('access_token', token, {
    httpOnly: true,
    sameSite: 'lax',
    secure,
    maxAge: 15 * 60 * 1000
  });

  res.json({ ok: true });
}

export async function logout(req, res) {
  res.clearCookie('access_token');
  res.json({ ok: true });
}

export async function me(req, res) {
  res.json({ user: req.user });
}
```

```js
export async function listTodos(req, res) {
  const [rows] = await pool.query(
    'SELECT id, title, done, created_at FROM todos WHERE user_id=? ORDER BY created_at DESC',
    [req.user.id]

  res.json({ todos: rows });
}

export async function addTodo(req, res) {
  const { title } = req.body;
  if (!title?.trim()) return res.status(400).json({ message: 'title required' });

  const id = crypto.randomUUID();
  await pool.query(
    'INSERT INTO todos (id, user_id, title, done) VALUES (?, ?, ?, ?)',
    [id, req.user.id, title.trim(), false]
  );

  res.json({ ok: true, id });
}

export async function toggleTodo(req, res) {
  const { id } = req.params;
  const { done } = req.body;

  await pool.query(
    'UPDATE todos SET done=? WHERE id=? AND user_id=?',
    [!!done, id, req.user.id]
  );

  res.json({ ok: true });
}

export async function deleteTodo(req, res) {
  const { id } = req.params;

  await pool.query(
    'DELETE FROM todos WHERE id=? AND user_id=?',
    [id, req.user.id]
  );

  res.json({ ok: true });
}
```

```js
async function api(path, options = {}) {
  const res = await fetch(path, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      ...(options.headers || {})
    },
    credentials: 'include'
  });

  const data = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(data.message || 'Request failed');
  return data;
}

const listEl = document.querySelector('#todoList');
const form = document.querySelector('#todoForm');
const input = document.querySelector('#todoInput');
const logoutBtn = document.querySelector('#logoutBtn');

let todos = [];

function render() {
  listEl.innerHTML = '';

  for (const t of todos) {
    const li = document.createElement('li');
    li.className = 'item' + (t.done ? ' done' : '');

    const left = document.createElement('div');
    left.style.display = 'flex';
    left.style.gap = '10px';

    const cb = document.createElement('input');
    cb.type = 'checkbox';
    cb.checked = !!t.done;
    cb.addEventListener('change', async () => {
      try {
        await api(`/api/todos/${t.id}`, {
          method: 'PATCH',
          body: JSON.stringify({ done: cb.checked })
        });
        t.done = cb.checked;
        render();
      } catch (e) {
        alert(e.message);
        cb.checked = !cb.checked;
      }
    });

    const title = document.createElement('span');
    title.className = 'title';
    title.textContent = t.title;

    left.appendChild(cb);
    left.appendChild(title);

    const actions = document.createElement('div');
    actions.className = 'actions';

    const del = document.createElement('button');
    del.textContent = 'حذف';
    del.addEventListener('click', async () => {
      if (!confirm('حذف شود؟')) return;
      try {
        await api(`/api/todos/${t.id}`, { method: 'DELETE' });
        todos = todos.filter(x => x.id !== t.id);
        render();
      } catch (e) {
        alert(e.message);
      }
    });

    actions.appendChild(del);

    li.appendChild(left);
    li.appendChild(actions);
    listEl.appendChild(li);
  }
}

async function load() {
  try {
    await api('/api/auth/me');
    const data = await api('/api/todos');
    todos = data.todos;
    render();
  } catch {
    location.href = 'login.html';
  }
}

form.addEventListener('submit', async (e) => {
  e.preventDefault();
  const title = input.value.trim();
  if (!title) return;

  try {
    const res = await api('/api/todos', {
      method: 'POST',
      body: JSON.stringify({ title })
    });

    todos.unshift({ id: res.id, title, done: false });
    input.value = '';
    render();

    alert(e.message);
  }
});
```
